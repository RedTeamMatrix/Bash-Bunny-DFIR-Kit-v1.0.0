ATTACKMODE HID STORAGE

# -----------------------------
# DFIR Collector payload (LED states + timing)
# LED Y FAST   -> launching (Run box + typing)
# LED M SLOW   -> waiting for UAC approval
# LED B SLOW   -> collecting (active)
# LED G SOLID  -> finished
# LED R SOLID  -> error (not implemented automatically)
# -----------------------------

# 1) Launching - yellow fast blink while HID types the launcher
LED Y FAST

# Open Run and launch elevated PowerShell that runs the collector script
QUACK GUI r
QUACK DELAY 200
QUACK STRING powershell -NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -Command "Start-Process powershell -ArgumentList '-NoProfile -ExecutionPolicy Bypass -File \"D:\payloads\switch1\collector.ps1\"' -Verb RunAs -Wait"
QUACK ENTER

# 2) UAC waiting - magenta slow blink
# Give operator time to click "Yes" on UAC (10 seconds)
LED M SLOW
Q DELAY 10000

# 3) Collecting - blue slow blink
# Wait long enough for collection to finish on most hosts (default 5 minutes)
LED B SLOW
Q DELAY 300000

# 4) Finished - green solid
LED G SOLID
# short visible hold so user notices green
Q DELAY 3000

# turn LED off after short pause
LED OFF
